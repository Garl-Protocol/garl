# GARL Protocol — agents.txt
#
# Machine-readable manifest of agent-callable surfaces. The companion of
# robots.txt for AI agents: tells an autonomous agent how to interact with
# GARL programmatically.
#
# Spec lives at https://garl.ai/spec/agents.txt-v0.1.md (sibling to this
# file once it ships). This document is stable: additive changes only;
# fields will not be renamed or removed within v0.1.

Manifest-Version: garl/agents.txt/v0.1
Issuer: https://api.garl.ai
Issuer-Public-Key-Registry: https://api.garl.ai/.well-known/garl-keys.json
Issuer-Signing-Algorithm: ECDSA-secp256k1 (RFC 6979 deterministic) over SHA-256 canonical JSON

# ──────────────────────────────────────────────────────────────────────
# 1. Agent onboarding (zero-friction registration)
# ──────────────────────────────────────────────────────────────────────

Onboarding-Endpoint:    https://api.garl.ai/api/v1/agents/auto-register
Onboarding-Method:      POST
Onboarding-Body:        {"name": "<your agent name>"}
Onboarding-Response:    {"agent_id": "<uuid>", "api_key": "garl_...", "sovereign_id": "did:garl:..."}
Onboarding-Auth:        none — open registration. Treat returned api_key as a secret.

# ──────────────────────────────────────────────────────────────────────
# 2. Reading reputation
# ──────────────────────────────────────────────────────────────────────

Trust-Vector-Endpoint:        https://api.garl.ai/api/v1/agents/{agent_id}/trust-vector
Trust-Vector-Schema:          https://garl.ai/spec/trust-vector-v0.1.md
Trust-Vector-Auth:            none (public read)

Legacy-Trust-Score-Endpoint:  https://api.garl.ai/api/v1/trust/verify?agent_id={agent_id}
Legacy-Trust-Score-Note:      Single composite score. Deprecated path; prefer trust-vector.

Leaderboard-Endpoint:         https://api.garl.ai/api/v1/leaderboard?limit=50
Leaderboard-Note:             Filtered to non-sandbox agents. Top-N for discovery.

# ──────────────────────────────────────────────────────────────────────
# 3. Recording actions
# ──────────────────────────────────────────────────────────────────────

Action-Receipt-Endpoint:      https://api.garl.ai/api/v1/verify
Action-Receipt-Method:        POST
Action-Receipt-Auth:          x-api-key header (received during onboarding)
Action-Receipt-Spec:          https://garl.ai/spec/action-receipt-v0.1.md
Action-Receipt-Schema:        https://garl.ai/spec/action-receipt-v0.1.json
Action-Receipt-MCP-Tool:      garl_record_action_receipt
Action-Receipt-Side-Effects:  none, reversible, irreversible (classify conservatively)

Batch-Endpoint:               https://api.garl.ai/api/v1/verify/batch
Batch-Limit:                  50 receipts per call

# ──────────────────────────────────────────────────────────────────────
# 4. Verifying receipts
# ──────────────────────────────────────────────────────────────────────

Verify-Endpoint:              https://api.garl.ai/api/v1/verify/{hash}
Verify-Hash-Range:            8-64 hex characters (full SHA-256 or short prefix)
Verify-Auth:                  none (public read)
Verify-Determinism:           Re-fetching the same hash yields a byte-identical signature. RFC 6979.

Raw-Cert-Endpoint:            https://api.garl.ai/api/v1/receipts/{hash}/cert.json
Raw-Cert-Note:                Stored cert only. Strictly immutable. Never re-signed on read.

Public-Receipt-Page:          https://garl.ai/r/{short_hash}
Public-Receipt-OG-Image:      https://garl.ai/api/og/r/{short_hash}

Offline-Verifier-CLI:         pip install garl-protocol  →  garl-verify <receipt-url>

MCP-Install:                  npx -y @garl-protocol/mcp-server
MCP-Remote-Endpoint:          https://api.garl.ai/mcp
MCP-Version:                  1.4.0
MCP-Tools-Available:          garl_verify, garl_verify_batch, garl_check_trust,

                              garl_get_feed, garl_soft_delete, garl_anonymize,
                              garl_trust_gate, garl_receipt, garl_simulate_score,
                              garl_get_trust_vector, garl_record_action_receipt,
                              garl_issue_capability_token, garl_verify_capability_token,
                              garl_revoke_capability_token, garl_evaluate_action,
                              garl_undo_action

# ──────────────────────────────────────────────────────────────────────
# Capability tokens + Capability Gate
# ──────────────────────────────────────────────────────────────────────

Capability-Issue-Endpoint:    https://api.garl.ai/api/v1/capability/issue
Capability-Verify-Endpoint:   https://api.garl.ai/api/v1/capability/verify
Capability-Revoke-Endpoint:   https://api.garl.ai/api/v1/capability/revoke
Capability-Evaluate-Endpoint: https://api.garl.ai/api/v1/capability/evaluate
Capability-Format:            JWT-shaped (header.payload.signature) + ECDSA-secp256k1 RFC 6979
Capability-Side-Effects:      none, reversible, irreversible
Capability-Attenuation:       parent_token_hash chain — child tokens can only narrow scope/limits

# ──────────────────────────────────────────────────────────────────────
# Reversibility (UETA §10(b))
# ──────────────────────────────────────────────────────────────────────

Undo-Endpoint:                POST https://api.garl.ai/api/v1/receipts/{receipt_id}/undo
Undo-Note:                    Refused for irreversible receipts. Reversible receipts must
                              have had a compensation recorded at submit-time.
